# nginx/nginx.conf
events {
    worker_connections 1024;
}

http {
    upstream booking_api {
        server booking-api:8000;
    }
    
    upstream ai_agent {
        server ai-agent:8000;
    }
    
    upstream n8n_service {
        server n8n:5678;
    }

    # Rate limiting
    limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s;
    limit_req_zone $binary_remote_addr zone=booking:10m rate=2r/s;

    server {
        listen 80;
        server_name instylehairboutique.co.za;
        
        # Redirect HTTP to HTTPS
        return 301 https://$server_name$request_uri;
    }

    server {
        listen 443 ssl http2;
        server_name instylehairboutique.co.za;

        # SSL Configuration
        ssl_certificate /etc/nginx/ssl/instylehairboutique.co.za.crt;
        ssl_certificate_key /etc/nginx/ssl/instylehairboutique.co.za.key;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512;
        ssl_prefer_server_ciphers off;

        # Security headers
        add_header X-Frame-Options DENY;
        add_header X-Content-Type-Options nosniff;
        add_header X-XSS-Protection "1; mode=block";
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains";

        # API routes
        location /api/book {
            limit_req zone=booking burst=5 nodelay;
            proxy_pass http://booking_api/book;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        location /api/ {
            limit_req zone=api burst=20 nodelay;
            proxy_pass http://booking_api/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # AI Agent routes
        location /ai/ {
            limit_req zone=api burst=10 nodelay;
            proxy_pass http://ai_agent/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # N8N webhooks
        location /webhook/ {
            proxy_pass http://n8n_service/webhook/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # Static files and Next.js app (served by Vercel)
        location / {
            proxy_pass https://appointmentbooking-instyle.vercel.app;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # Fallback for client-side routing
            proxy_intercept_errors on;
            error_page 404 = @fallback;
        }

        location @fallback {
            proxy_pass https://appointmentbooking-instyle.vercel.app/;
        }
    }
}

# deploy.sh - Production deployment script
#!/bin/bash
set -e

echo "🚀 Starting Instyle Hair Boutique deployment..."

# Check prerequisites
command -v docker >/dev/null 2>&1 || { echo "Docker is required but not installed. Aborting." >&2; exit 1; }
command -v docker-compose >/dev/null 2>&1 || { echo "Docker Compose is required but not installed. Aborting." >&2; exit 1; }

# Load environment variables
if [ -f .env ]; then
    source .env
else
    echo "❌ .env file not found! Please create one based on .env.example"
    exit 1
fi

# Create necessary directories
mkdir -p nginx/ssl
mkdir -p n8n/workflows
mkdir -p logs

# Build and start services
echo "📦 Building and starting services..."
docker-compose up --build -d

# Wait for services to be healthy
echo "⏳ Waiting for services to start..."
sleep 30

# Run database migrations
echo "🗄️ Running database migrations..."
docker-compose exec booking-api python -c "
import asyncio
import asyncpg
import os

async def migrate():
    conn = await asyncpg.connect(os.getenv('DATABASE_URL'))
    
    # Check if tables exist
    result = await conn.fetch('''
        SELECT table_name FROM information_schema.tables 
        WHERE table_schema = 'public'
    ''')
    
    if not result:
        print('Running initial migration...')
        with open('/app/migrations/init.sql', 'r') as f:
            await conn.execute(f.read())
        print('Migration complete!')
    else:
        print('Database already migrated.')
    
    await conn.close()

asyncio.run(migrate())
"

# Health checks
echo "🏥 Running health checks..."
services=("booking-api:8000/health" "ai-agent:8000/health")

for service in "${services[@]}"; do
    url="http://localhost:${service}"
    if curl -f -s "$url" > /dev/null; then
        echo "✅ $service is healthy"
    else
        echo "❌ $service is not responding"
        exit 1
    fi
done

# Test booking flow
echo "🧪 Testing booking flow..."
booking_response=$(curl -s -X POST http://localhost:8000/book \
  -H "Content-Type: application/json" \
  -d '{
    "tenant_id": "'$INSTYLE_TENANT_ID'",
    "service_id": "test-service",
    "client_name": "Test Client",
    "client_phone": "+27821234567",
    "start_time": "'$(date -d '+1 day' --iso-8601=seconds)'",
    "consent_popia": true
  }' || echo "FAILED")

if echo "$booking_response" | grep -q "id"; then
    echo "✅ Booking API is working"
else
    echo "❌ Booking API test failed: $booking_response"
fi

# Display deployment info
echo ""
echo "🎉 Deployment complete!"
echo ""
echo "📊 Service Status:"
echo "├── Database: Running on :5432"
echo "├── Booking API: Running on :8000"
echo "├── AI Agent: Running on :8001"
echo "├── WhatsApp Bot: Running on :5678"
echo "└── Nginx Proxy: Running on :80/:443"
echo ""
echo "🌐 URLs:"
echo "├── Website: https://instylehairboutique.co.za"
echo "├── API: https://instylehairboutique.co.za/api"
echo "├── Dashboard: https://instylehairboutique.co.za/dashboard"
echo "└── N8N Admin: http://localhost:5678 (admin/$(echo $N8N_PASSWORD))"
